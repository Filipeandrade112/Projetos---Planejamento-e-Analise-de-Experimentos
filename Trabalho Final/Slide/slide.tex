\begin{frame}{Modelagem Estatística em Blocos e Hipóteses}
    \footnotesize
    Para isolar a variabilidade estrutural entre os \textbf{5 circuitos de RF} e avaliar apenas a diferença entre os \textbf{3 métodos}, adota-se o modelo linear aditivo em blocos completos (RCBD):

    \vspace{0.5cm}
    \centering
    % Fluxograma da decomposição do modelo e soma de quadrados
    \begin{tikzpicture}[scale=0.85, transform shape,
        eqbox/.style={draw=meuvermelho, line width=1pt, rounded corners=4pt, fill=gray!6, inner sep=5.5pt, align=center},
        partbox/.style={draw, line width=0.9pt, rounded corners=3pt, minimum width=4.1cm, minimum height=1.5cm, align=center, inner sep=4pt},
        arr/.style={-{Stealth[length=2.3mm]}, line width=1pt, draw=meuvermelho!85!black}
    ]
        % Equação principal no topo
        \node[eqbox, minimum width=13.3cm] (mod) at (0, 1.5) {
            \small $y_{ij} = \mu + \tau_i + \beta_j + \epsilon_{ij}, \qquad \epsilon_{ij} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, \sigma^2)$\\[3pt]
            \scriptsize $\mu$: Média global \quad$|$\quad $\tau_i$: Efeito do Método ($i=1,2,3$) \quad$|$\quad $\beta_j$: Efeito do Circuito/Bloco ($j=1,\dots,5$)
        };

        % Três caixas da partição da variância
        \node[partbox, draw=corHFSS!85!black, fill=verdeclaro] (ssm) at (-4.6, -0.5) {
            \scriptsize \textbf{Efeito dos Métodos ($\tau_i$)}\\[2pt]
            \footnotesize $SS_{\text{Métodos}}$ \scriptsize ($a-1 = 2\text{ GL}$)\\[2pt]
            \tiny Variabilidade entre HFSS, Tidy3D e Analítico
        };

        \node[partbox, draw=corBloco!85!black, fill=corBloco!10] (ssb) at (0, -0.5) {
            \scriptsize \textbf{Efeito dos Circuitos ($\beta_j$)}\\[2pt]
            \footnotesize $SS_{\text{Blocos}}$ \scriptsize ($b-1 = 4\text{ GL}$)\\[2pt]
            \tiny Variabilidade isolada dos 5 dispositivos
        };

        \node[partbox, draw=black!60, fill=gray!12] (sse) at (4.6, -0.5) {
            \scriptsize \textbf{Erro Residual ($\epsilon_{ij}$)}\\[2pt]
            \footnotesize $SS_{\text{Erro}}$ \scriptsize ($(a-1)(b-1) = 8\text{ GL}$)\\[2pt]
            \tiny Variância residual livre do efeito de bloco
        };

        \draw[arr] (ssm.north |- mod.south) -- (ssm.north);
        \draw[arr] (ssb.north |- mod.south) -- (ssb.north);
        \draw[arr] (sse.north |- mod.south) -- (sse.north);

        % Hipóteses na base
        \node[draw=meuvermelho, line width=0.9pt, fill=azulclaro!50, rounded corners=4pt, minimum width=13.3cm, inner sep=5pt, align=center] (hip) at (0, -2.25) {
            \footnotesize \textbf{Hipóteses do Teste ANOVA em Blocos ($F_0 = MS_{\text{Métodos}} / MS_{\text{Erro}}$):}\\[2pt]
            \footnotesize $H_0: \tau_{\text{HFSS}} = \tau_{\text{Tidy3D}} = \tau_{\text{Analítico}} = 0$ \quad (Métodos equivalentes) \qquad vs. \qquad $H_1: \exists\, \tau_i \neq 0$ \quad (Há diferença entre os métodos)
        };

        \draw[arr] (ssm.south) -- (ssm.south |- hip.north);
        \draw[arr] (sse.south) -- (sse.south |- hip.north);
    \end{tikzpicture}
\end{frame}
